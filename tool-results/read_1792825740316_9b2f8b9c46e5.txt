   115→const STORAGE_KEY = 'kiyan-state-v1';
   116→
   117→interface SavedState {
   118→  score: number;
   119→  bestStreak: number;
   120→  solved: number;
   121→}
   122→
   123→// Cached default snapshot for SSR (must be stable to avoid useSyncExternalStore infinite loop)
   124→const DEFAULT_SAVED_STATE: SavedState = { score: 0, bestStreak: 0, solved: 0 };
   125→
   126→// Module-level cache of the last seen snapshot. useSyncExternalStore requires
   127→// getSnapshot to return the SAME reference if nothing changed; otherwise it
   128→// re-renders infinitely. We compute the snapshot once and only swap the
   129→// reference when the underlying localStorage value actually changes.
   130→let cachedSavedState: SavedState = DEFAULT_SAVED_STATE;
   131→
   132→function readSavedStateRaw(): SavedState {
   133→  if (typeof window === 'undefined') return DEFAULT_SAVED_STATE;
   134→  try {
   135→    const raw = window.localStorage.getItem(STORAGE_KEY);
   136→    if (!raw) return DEFAULT_SAVED_STATE;
   137→    const parsed = JSON.parse(raw) as Partial<SavedState>;
   138→    return {
   139→      score: parsed.score ?? 0,
   140→      bestStreak: parsed.bestStreak ?? 0,
   141→      solved: parsed.solved ?? 0,
   142→    };
   143→  } catch {
   144→    return DEFAULT_SAVED_STATE;
   145→  }
   146→}
   147→
   148→/**
   149→ * Returns the cached snapshot of saved state, swapping the cache when the
   150→ * underlying values change. This stable-reference pattern keeps
   151→ * useSyncExternalStore happy.
   152→ */
   153→function getSavedSnapshot(): SavedState {
   154→  if (typeof window === 'undefined') return DEFAULT_SAVED_STATE;
   155→  const next = readSavedStateRaw();
   156→  if (
   157→    cachedSavedState.score !== next.score ||
   158→    cachedSavedState.bestStreak !== next.bestStreak ||
   159→    cachedSavedState.solved !== next.solved
   160→  ) {
   161→    cachedSavedState = next;
   162→  }
   163→  return cachedSavedState;
   164→}

   166→function RainbowText({ text, className }: { text: string; className?: string }) {
   167→  const chars = text.split('');
   168→  return (
   169→    <span className={className} aria-label={text}>
   170→      {chars.map((c, i) => (
   171→        <span
   172→          key={i}
   173→          className={c.trim() === '' ? '' : RAINBOW_COLORS[i % RAINBOW_COLORS.length]}
   174→          aria-hidden
   175→        >
   176→          {c}
   177→        </span>
   178→      ))}
   179→    </span>
   180→  );
   181→}

   183→export default function Home() {
   184→  const [listening, setListening] = useState(false);
   185→  const [interim, setInterim] = useState('');
   186→  const [display, setDisplay] = useState<DisplayState | null>(null);
   187→  const [mood, setMood] = useState<MascotMood>('idle');
   188→  // Score, bestStreak, solved are the source of truth from localStorage
   189→  // (SSR-safe via useSyncExternalStore). Writes go through `updateSaved`.
   190→  const subscribeSaved = useCallback((cb: () => void) => {
   191→    if (typeof window === 'undefined') return () => {};
   192→    window.addEventListener('storage', cb);
   193→    window.addEventListener('kiyan-state-updated', cb);
   194→    return () => {
   195→      window.removeEventListener('storage', cb);
   196→      window.removeEventListener('kiyan-state-updated', cb);
   197→    };
   198→  }, []);
   199→  const savedState = useSyncExternalStore(
   200→    subscribeSaved,
   201→    getSavedSnapshot,
   202→    () => DEFAULT_SAVED_STATE
   203→  );
   204→  const { score, bestStreak, solved } = savedState;
   205→  // Streak is session-only, not persisted.
   206→  const [streak, setStreak] = useState(0);
   207→  const [confettiRun, setConfettiRun] = useState(0);
   208→  const [muted, setMuted] = useState(false);
   209→  const [showHelp, setShowHelp] = useState(false);
   210→  const [showLevelUp, setShowLevelUp] = useState<string | null>(null);
   211→  const [hintIndex, setHintIndex] = useState(0);
   212→  const recRef = useRef<SpeechRecognition | null>(null);
   213→  // Speech is queued here during a voice session and spoken only AFTER the
   214→  // recognition session has fully ended. Speaking while the mic is open makes
   215→  // the recognizer transcribe the app's own voice — a major source of
   216→  // "recognition not working" complaints.
   217→  const pendingSpeechRef = useRef<string | null>(null);
   218→  // Auto-returns the result card to the ready state so the app is always
   219→  // fresh for the next command.
   220→  const autoClearRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   221→  // Set when the user deliberately stops listening, so the trailing
   222→  // 'no-speech' error is suppressed instead of popping an error card.
   223→  const stopRequestedRef = useRef(false);
   224→
   225→  // Helper that writes to localStorage and notifies subscribers.
   226→  const updateSaved = useCallback((updater: (prev: SavedState) => SavedState) => {
   227→    if (typeof window === 'undefined') return;
   228→    try {
   229→      const current = readSavedStateRaw();
   230→      const next = updater(current);
   231→      window.localStorage.setItem(STORAGE_KEY, JSON.stringify(next));
   232→      window.dispatchEvent(new Event('kiyan-state-updated'));
   233→    } catch {
   234→      /* ignore */
   235→    }
   236→  }, []);
   237→
   238→  // Detect support for SpeechRecognition (SSR-safe via useSyncExternalStore)
   239→  const supported = useSyncExternalStore(
   240→    () => () => {},
   241→    () =>
   242→      typeof window !== 'undefined' &&
   243→      !!(window.SpeechRecognition || window.webkitSpeechRecognition),
   244→    () => false
   245→  );
   246→
   247→  // Preload TTS voices (Chrome often loads them asynchronously)
   248→  useEffect(() => {
   249→    if (typeof window === 'undefined' || !('speechSynthesis' in window)) return;
   250→    window.speechSynthesis.getVoices();
   251→    const handler = () => window.speechSynthesis.getVoices();
   252→    window.speechSynthesis.addEventListener?.('voiceschanged', handler);
   253→    return () => window.speechSynthesis.removeEventListener?.('voiceschanged', handler);
   254→  }, []);
   255→
   256→  // Rotate the hint every few seconds when idle
   257→  useEffect(() => {
   258→    if (mood !== 'idle') return;
   259→    const t = setInterval(() => {
   260→      setHintIndex((i) => (i + 1) % EXAMPLE_PHRASES.length);
   261→    }, 3500);
   262→    return () => clearInterval(t);
   263→  }, [mood]);
   264→
   265→  // Cancel any pending auto-clear timer.
   266→  const cancelAutoClear = useCallback(() => {
   267→    if (autoClearRef.current) {
   268→      clearTimeout(autoClearRef.current);
   269→      autoClearRef.current = null;
   270→    }
   271→  }, []);
   272→
   273→  // Full reset so the next voice command starts from a clean slate.
   274→  const resetForNewCommand = useCallback(() => {
   275→    cancelAutoClear();
   276→    setDisplay(null);
   277→    setInterim('');
   278→    setMood('idle');
   279→  }, [cancelAutoClear]);
   280→
   281→  // Show a friendly error card, then auto-refresh to the ready state.
   282→  const showErrorCard = useCallback(
   283→    (message: string, error?: string) => {
   284→      cancelAutoClear();
   285→      setMood('oops');
   286→      setDisplay({ expression: '', result: null, message, transcript: '', error });
   287→      autoClearRef.current = setTimeout(() => {
   288→        setDisplay(null);
   289→        setInterim('');
   290→        setMood((m) => (m === 'oops' ? 'idle' : m));
   291→      }, 6000);
   292→    },
   293→    [cancelAutoClear]
   294→  );
   295→
   296→  const handleResult = useCallback(
   297→    (parsed: ParseResult, transcript: string, fromVoice: boolean) => {
   298→      cancelAutoClear();
   299→
   300→      // Queue speech (voice path) or speak immediately (button path).
   301→      const say = (text: string) => {
   302→        if (muted) return;
   303→        if (fromVoice) {
   304→          pendingSpeechRef.current = pendingSpeechRef.current
   305→            ? `${pendingSpeechRef.current} ${text}`
   306→            : text;
   307→        } else {
   308→          speak(text, { pitch: 1.3, rate: 0.95 });
   309→        }
   310→      };
   311→
   312→      if (parsed.result === null || parsed.expression === '') {
   313→        setStreak(0);
   314→        showErrorCard(parsed.error ?? "I didn't hear that!", parsed.error);
   315→        say(parsed.error ?? 'Oops! Try again, buddy!');
   316→        return;
   317→      }
   318→
   319→      const message = ENCOURAGEMENTS[Math.floor(Math.random() * ENCOURAGEMENTS.length)];
   320→      setDisplay({
   321→        expression: parsed.expression,
   322→        result: parsed.result,
   323→        message,
   324→        transcript,
   325→      });
   326→      setMood('happy');
   327→      setConfettiRun((r) => r + 1);
   328→
   329→      // Speak the result (queued until the mic is off for voice input).
   330→      say(spokenResultText(parsed));
   331→
   332→      // Gamification: +10 per correct solve, +streak bonus.
   333→      // Persist score / bestStreak / solved via localStorage.
   334→      const newStreak = streak + 1;
   335→      setStreak(newStreak);
   336→      updateSaved((s) => ({
   337→        score: s.score + 10 + streak,
   338→        bestStreak: Math.max(s.bestStreak, newStreak),
   339→        solved: s.solved + 1,
   340→      }));
   341→      // Level-up celebration every 5 streak
   342→      if (newStreak > 0 && newStreak % 5 === 0) {
   343→        setShowLevelUp(`Streak ${newStreak}! 🔥`);
   344→        say(`Wow! Streak of ${newStreak}! You are a math star!`);
   345→        setTimeout(() => setShowLevelUp(null), 2500);
   346→      }
   347→
   348→      // Auto-refresh back to the ready state shortly after the celebration,
   349→      // so the next command always starts clean.
   350→      autoClearRef.current = setTimeout(() => {
   351→        setDisplay(null);
   352→        setInterim('');
   353→        setMood((m) => (m === 'happy' ? 'idle' : m));
   354→      }, 8000);
   355→
   356→      // Return to idle after a beat
   357→      setTimeout(() => {
   358→        setMood((m) => (m === 'happy' ? 'idle' : m));
   359→      }, 2200);
   360→    },
   361→    [streak, muted, updateSaved, cancelAutoClear, showErrorCard]
   362→  );
   363→
   364→  const startListening = useCallback(() => {
   365→    if (typeof window === 'undefined') return;
   366→    const ctor = window.SpeechRecognition || window.webkitSpeechRecognition;
   367→    if (!ctor) return;
   368→
   369→    // Stop any existing instance
   370→    if (recRef.current) {
   371→      try {
   372→        recRef.current.abort();
   373→      } catch {
   374→        /* ignore */
   375→      }
   376→      recRef.current = null;
   377→    }
   378→
   379→    const rec = new ctor();
   380→    rec.lang = 'en-US';
   381→    rec.continuous = false;
   382→    rec.interimResults = true;
   383→    rec.maxAlternatives = 3;
   384→
   385→    rec.onstart = () => {
   386→      stopRequestedRef.current = false;
   387→      pendingSpeechRef.current = null;
   388→      setListening(true);
   389→      setMood('listening');
   390→      setInterim('');
   391→      // Clear the previous result so the "Listening…" card is always visible.
   392→      setDisplay(null);
   393→      // NOTE: no TTS here on purpose — speaking while the mic is open makes
   394→      // the recognizer hear the app's own voice.
   395→    };
   396→
   397→    rec.onresult = (event: SpeechRecognitionEvent) => {
   398→      let interimText = '';
   399→      let finalEvent: SpeechRecognitionResult | null = null;
   400→      for (let i = event.resultIndex; i < event.results.length; i++) {
   401→        const r = event.results[i];
   402→        if (r.isFinal && !finalEvent) finalEvent = r;
   403→        else if (!r.isFinal) interimText += r[0].transcript;
   404→      }
   405→      if (interimText) setInterim(interimText);
   406→      if (finalEvent) {
   407→        setInterim('');
   408→        const transcript = finalEvent[0].transcript.trim();
   409→        // Try each alternative until one parses cleanly — this makes
   410→        // recognition far more forgiving of misheard words.
   411→        let parsed = parseSpokenMath(transcript);
   412→        for (let a = 1; a < finalEvent.length; a++) {
   413→          if (parsed.result !== null) break;
   414→          const altParsed = parseSpokenMath(finalEvent[a].transcript.trim());
   415→          if (altParsed.result !== null) {
   416→            parsed = { ...altParsed, spokenInput: transcript };
   417→          }
   418→        }
   419→        handleResult(parsed, transcript, true);
   420→      }
   421→    };
   422→
   423→    rec.onerror = (event: SpeechRecognitionErrorEvent) => {
   424→      setListening(false);
   425→      if (event.error === 'aborted') {
   426→        // User cancelled — UI already reset, stay quiet.
   427→        return;
   428→      }
   429→      if (stopRequestedRef.current && event.error === 'no-speech') {
   430→        // Deliberate stop with nothing said — keep the clean refreshed state.
   431→        return;
   432→      }
   433→      if (event.error === 'not-allowed' || event.error === 'service-not-allowed') {
   434→        showErrorCard(
   435→          'Microphone is blocked! Ask a grown-up to open this app in a new browser tab, then allow the mic.',
   436→          'mic-blocked'
   437→        );
   438→      } else if (event.error === 'no-speech') {
   439→        showErrorCard("I didn't hear anything — tap the mic and try again!");
   440→      } else if (event.error === 'network') {
   441→        showErrorCard(
   442→          'The speech service is not reachable right now. Try again, or use the buttons below!'
   443→        );
   444→      } else {
   445→        showErrorCard(`Oops: ${event.error}. Try again!`);
   446→      }
   447→    };
   448→
   449→    rec.onend = () => {
   450→      setListening(false);
   451→      stopRequestedRef.current = false;
   452→      setMood((m) => (m === 'listening' ? 'idle' : m));
   453→      // The mic is fully off now — speak anything queued during the session.
   454→      const queued = pendingSpeechRef.current;
   455→      pendingSpeechRef.current = null;
   456→      if (queued) {
   457→        setTimeout(() => speak(queued, { pitch: 1.3, rate: 0.95 }), 250);
   458→      }
   459→    };
   460→
   461→    try {
   462→      rec.start();
   463→      recRef.current = rec;
   464→    } catch {
   465→      // Some browsers throw if start() is called too rapidly
   466→      setListening(false);
   467→    }
   468→  }, [handleResult, cancelAutoClear, showErrorCard]);
   469→
   470→  const toggleMic = useCallback(() => {
   471→    if (listening) {
   472→      // STOP pressed — finish the session and fully refresh for a new command.
   473→      stopRequestedRef.current = true;
   474→      if (recRef.current) {
   475→        try {
   476→          // stop() politely finalises any speech already heard (so nothing
   477→          // the kid said is lost); the UI below resets immediately.
   478→          recRef.current.stop();
   479→        } catch {
   480→          /* ignore */
   481→        }
   482→        recRef.current = null;
   483→      }
   484→      pendingSpeechRef.current = null;
   485→      resetForNewCommand();
   486→      setListening(false);
   487→    } else {
   488→      // START pressed — clear any previous result, then listen.
   489→      resetForNewCommand();
   490→      startListening();
   491→    }
   492→  }, [listening, startListening, resetForNewCommand]);
   493→
   494→  // Quick-example buttons (mouse/keyboard fallback)
   495→  const tryExample = useCallback(
   496→    (phrase: string) => {
   497→      // If a voice session is somehow still open, close it quietly first.
   498→      if (listening && recRef.current) {
   499→        stopRequestedRef.current = true;
   500→        try {
   501→          recRef.current.abort();
   502→        } catch {
   503→          /* ignore */
   504→        }
   505→        recRef.current = null;
   506→        setListening(false);
   507→      }
   508→      pendingSpeechRef.current = null;
   509→      const parsed = parseSpokenMath(phrase);
   510→      handleResult(parsed, phrase, false);
   511→    },
   512→    [listening, handleResult]
   513→  );
   514→
   515→  const resetProgress = useCallback(() => {
   516→    setStreak(0);
   517→    setDisplay(null);
   518→    setMood('idle');
   519→    if (typeof window !== 'undefined') {
   520→      try {
   521→        window.localStorage.removeItem(STORAGE_KEY);
   522→        window.dispatchEvent(new Event('kiyan-state-updated'));
   523→      } catch {
   524→        /* ignore */
   525→      }
   526→    }
   527→  }, []);
   528→
   529→  const toggleMute = useCallback(() => {
   530→    setMuted((m) => {
   531→      const next = !m;
   532→      if (next && typeof window !== 'undefined' && 'speechSynthesis' in window) {
   533→        window.speechSynthesis.cancel();
   534→      }
   535→      return next;
   536→    });
   537→  }, []);
   538→
   539→  const micDisabled = supported === false;
   540→  const currentHint = EXAMPLE_PHRASES[hintIndex];
   541→
   542→  return (
   543→    <div className="min-h-screen relative overflow-hidden bg-gradient-to-br from-amber-200 via-orange-200 to-pink-200">
   544→      {/* Decorative floating shapes */}
   545→      <FloatingDecorations />
   546→
   547→      <Confetti run={confettiRun} />
   548→
   549→      {/* Level-up banner */}
   550→      <AnimatePresence>
   551→        {showLevelUp && (
   552→          <motion.div
   553→            initial={{ opacity: 0, scale: 0.6, y: -40 }}
   554→            animate={{ opacity: 1, scale: 1, y: 0 }}
   555→            exit={{ opacity: 0, scale: 0.6, y: -40 }}
   556→            className="fixed top-6 left-1/2 -translate-x-1/2 z-40 bg-gradient-to-r from-orange-500 to-pink-500 text-white px-6 py-3 rounded-full text-xl md:text-2xl font-black shadow-2xl border-4 border-white"
   557→          >
   558→            {showLevelUp}
   559→          </motion.div>
   560→        )}
   561→      </AnimatePresence>
   562→
   563→      <main className="relative z-10 flex min-h-screen flex-col">
   564→        {/* Header */}
   565→        <header className="px-4 pt-6 pb-2 md:pt-8 md:px-8">
   566→          <div className="mx-auto max-w-5xl flex items-center justify-between gap-4 flex-wrap">
   567→            <div>
   568→              <h1 className="text-5xl md:text-7xl font-black tracking-tight leading-none drop-shadow-sm">
   569→                <RainbowText text="KIYAN" />
   570→              </h1>
   571→              <p className="text-base md:text-xl font-bold text-purple-700 mt-1">
   572→                Voice Math Adventure
   573→              </p>
   574→            </div>
   575→
   576→            <ScorePanel
   577→              score={score}
   578→              streak={streak}
   579→              bestStreak={bestStreak}
   580→              solved={solved}
   581→              onReset={resetProgress}
   582→              muted={muted}
   583→              onToggleMute={toggleMute}

   844→function EquationText({ expression, result }: { expression: string; result: number }) {

   850→  }, [expression]);

   857→        const isOp = ['+', '−', '×', '÷', '='].includes(t);

   865→            }
   866→          >
   867→            {t}

   872→      <motion.span
   873→        initial={{ scale: 1.4 }}

  1017→// Deterministic PRNG (mulberry32) so the floating decorations are computed
  1018→// identically on the server and the client — no hydration mismatch, and
  1019→// no need to setState-after-mount.
  1020→function mulberry32(seed: number) {
  1021→  let a = seed >>> 0;
  1022→  return () => {
  1023→    a = (a + 0x6d2b79f5) >>> 0;
  1024→    let t = a;
  1025→    t = Math.imul(t ^ (t >>> 15), t | 1);
  1026→    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
  1027→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  1028→  };
  1029→}

  1074→function spokenResultText(parsed: ParseResult): string {
  1075→  if (parsed.result === null) return '';
  1076→  const r = parsed.result;
  1077→  const rStr = Number.isInteger(r) ? numberToWords(r) : r.toFixed(2);
  1078→  return `${parsed.expression.replace(/×/g, 'times').replace(/÷/g, 'divided by').replace(/−/g, 'minus')} equals ${rStr}.`;
  1079→}
  1080→
  1081→const NUM_WORDS_UNDER_100: string[] = [
  1082→  'zero', 'one', 'two', 'three', 'four', 'five', 'six', 'seven', 'eight', 'nine',
  1083→  'ten', 'eleven', 'twelve', 'thirteen', 'fourteen', 'fifteen', 'sixteen',
  1084→  'seventeen', 'eighteen', 'nineteen',
  1085→];
  1086→const TENS: string[] = [
  1087→  '', '', 'twenty', 'thirty', 'forty', 'fifty', 'sixty', 'seventy', 'eighty', 'ninety',
  1088→];
  1089→
  1090→function numberToWords(n: number): string {
  1091→  if (n < 0) return 'negative ' + numberToWords(-n);
  1092→  if (n < 20) return NUM_WORDS_UNDER_100[n];
  1093→  if (n < 100) {
  1094→    const t = Math.floor(n / 10);
  1095→    const r = n % 10;
  1096→    return r === 0 ? TENS[t] : `${TENS[t]} ${NUM_WORDS_UNDER_100[r]}`;
  1097→  }
  1098→  if (n < 1000) {
  1099→    const h = Math.floor(n / 100);
  1100→    const r = n % 100;
  1101→    return r === 0 ? `${NUM_WORDS_UNDER_100[h]} hundred` : `${NUM_WORDS_UNDER_100[h]} hundred ${numberToWords(r)}`;
  1102→  }
  1103→  if (n < 1_000_000) {
  1104→    const th = Math.floor(n / 1000);
