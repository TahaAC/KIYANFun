   117→   117→interface SavedState {

   123→   123→// Cached default snapshot for SSR (must be stable to avoid useSyncExternalStore infinite loop)
   124→   124→const DEFAULT_SAVED_STATE: SavedState = { score: 0, bestStreak: 0, solved: 0 };
   125→   125→
   126→   126→// Module-level cache of the last seen snapshot. useSyncExternalStore requires
   127→   127→// getSnapshot to return the SAME reference if nothing changed; otherwise it
   128→   128→// re-renders infinitely. We compute the snapshot once and only swap the
   129→   129→// reference when the underlying localStorage value actually changes.
   130→   130→let cachedSavedState: SavedState = DEFAULT_SAVED_STATE;
   131→   131→
   132→   132→function readSavedStateRaw(): SavedState {
   133→   133→  if (typeof window === 'undefined') return DEFAULT_SAVED_STATE;
   134→   134→  try {
   135→   135→    const raw = window.localStorage.getItem(STORAGE_KEY);
   136→   136→    if (!raw) return DEFAULT_SAVED_STATE;
   137→   137→    const parsed = JSON.parse(raw) as Partial<SavedState>;
   138→   138→    return {
   139→   139→      score: parsed.score ?? 0,
   140→   140→      bestStreak: parsed.bestStreak ?? 0,
   141→   141→      solved: parsed.solved ?? 0,
   142→   142→    };
   143→   143→  } catch {
   144→   144→    return DEFAULT_SAVED_STATE;
   145→   145→  }
   146→   146→}
   147→   147→
   148→   148→/**
   149→   149→ * Returns the cached snapshot of saved state, swapping the cache when the
   150→   150→ * underlying values change. This stable-reference pattern keeps
   151→   151→ * useSyncExternalStore happy.
   152→   152→ */
   153→   153→function getSavedSnapshot(): SavedState {
   154→   154→  if (typeof window === 'undefined') return DEFAULT_SAVED_STATE;
   155→   155→  const next = readSavedStateRaw();
   156→   156→  if (
   157→   157→    cachedSavedState.score !== next.score ||
   158→   158→    cachedSavedState.bestStreak !== next.bestStreak ||
   159→   159→    cachedSavedState.solved !== next.solved
   160→   160→  ) {
   161→   161→    cachedSavedState = next;
   162→   162→  }
   163→   163→  return cachedSavedState;
   164→   164→}

   364→   364→  const startListening = useCallback(() => {
   365→   365→    if (typeof window === 'undefined') return;
   366→   366→    const ctor = window.SpeechRecognition || window.webkitSpeechRecognition;
   367→   367→    if (!ctor) return;

   369→   369→    // Stop any existing instance
   370→   370→    if (recRef.current) {

   373→   373→      } catch {
   374→   374→        /* ignore */
   375→   375→      }
   376→   376→      recRef.current = null;
   377→   377→    }
   378→   378→
   379→   379→    const rec = new ctor();
   380→   380→    rec.lang = 'en-US';
   381→   381→    rec.continuous = false;

   389→   389→      setMood('listening');

   391→   391→      // Clear the previous result so the "Listening…" card is always visible.
   392→   392→      setDisplay(null);
   393→   393→      // NOTE: no TTS here on purpose — speaking while the mic is open makes
   394→   394→      // the recognizer hear the app's own voice.
   395→   395→    };
   396→   396→
   397→   397→    rec.onresult = (event: SpeechRecognitionEvent) => {
   398→   398→      let interimText = '';
   399→   399→      let finalEvent: SpeechRecognitionResult | null = null;
   400→   400→      for (let i = event.resultIndex; i < event.results.length; i++) {
   401→   401→        const r = event.results[i];
   402→   402→        if (r.isFinal && !finalEvent) finalEvent = r;
   403→   403→        else if (!r.isFinal) interimText += r[0].transcript;
   404→   404→      }
   405→   405→      if (interimText) setInterim(interimText);
   406→   406→      if (finalEvent) {
   407→   407→        setInterim('');
   408→   408→        const transcript = finalEvent[0].transcript.trim();
   409→   409→        // Try each alternative until one parses cleanly — this makes
   410→   410→        // recognition far more forgiving of misheard words.
   411→   411→        let parsed = parseSpokenMath(transcript);
   412→   412→        for (let a = 1; a < finalEvent.length; a++) {
   413→   413→          if (parsed.result !== null) break;
   414→   414→          const altParsed = parseSpokenMath(finalEvent[a].transcript.trim());
   415→   415→          if (altParsed.result !== null) {
   416→   416→            parsed = { ...altParsed, spokenInput: transcript };
   417→   417→          }
   418→   418→        }
   419→   419→        handleResult(parsed, transcript, true);
   420→   420→      }
   421→   421→    };
   422→   422→
   423→   423→    rec.onerror = (event: SpeechRecognitionErrorEvent) => {
   424→   424→      setListening(false);
   425→   425→      if (event.error === 'aborted') {
   426→   426→        // User cancelled — UI already reset, stay quiet.
   427→   427→        return;
   428→   428→      }
   429→   429→      if (stopRequestedRef.current && event.error === 'no-speech') {
   430→   430→        // Deliberate stop with nothing said — keep the clean refreshed state.
   431→   431→        return;
   432→   432→      }
   433→   433→      if (event.error === 'not-allowed' || event.error === 'service-not-allowed') {
   434→   434→        showErrorCard(
   435→   435→          'Microphone is blocked! Ask a grown-up to open this app in a new browser tab, then allow the mic.',
   436→   436→          'mic-blocked'
   437→   437→        );
   438→   438→      } else if (event.error === 'no-speech') {
   439→   439→        showErrorCard("I didn't hear anything — tap the mic and try again!");
   440→   440→      } else if (event.error === 'network') {
   441→   441→        showErrorCard(
   442→   442→          'The speech service is not reachable right now. Try again, or use the buttons below!'
   443→   443→        );
   444→   444→      } else {
   445→   445→        showErrorCard(`Oops: ${event.error}. Try again!`);
   446→   446→      }
   447→   447→    };
   448→   448→
   449→   449→    rec.onend = () => {
   450→   450→      setListening(false);
   451→   451→      stopRequestedRef.current = false;
   452→   452→      setMood((m) => (m === 'listening' ? 'idle' : m));
   453→   453→      // The mic is fully off now — speak anything queued during the session.
   454→   454→      const queued = pendingSpeechRef.current;
   455→   455→      pendingSpeechRef.current = null;
   456→   456→      if (queued) {
   457→   457→        setTimeout(() => speak(queued, { pitch: 1.3, rate: 0.95 }), 250);
   458→   458→      }
   459→   459→    };
   460→   460→
   461→   461→    try {
   462→   462→      rec.start();
   463→   463→      recRef.current = rec;
   464→   464→    } catch {
   465→   465→      // Some browsers throw if start() is called too rapidly
   466→   466→      setListening(false);
   467→   467→    }
   468→   468→  }, [handleResult, cancelAutoClear, showErrorCard]);
   469→   469→
   470→   470→  const toggleMic = useCallback(() => {
   471→   471→    if (listening) {
   472→   472→      // STOP pressed — finish the session and fully refresh for a new command.
   473→   473→      stopRequestedRef.current = true;
   474→   474→      if (recRef.current) {
   475→   475→        try {
   476→   476→          // stop() politely finalises any speech already heard (so nothing
   477→   477→          // the kid said is lost); the UI below resets immediately.
   478→   478→          recRef.current.stop();
   479→   479→        } catch {

   844→   844→function EquationText({ expression, result }: { expression: string; result: number }) {
   845→   845→  // Split expression into tokens so we can color each number/operator.
   846→   846→  const tokens = useMemo(() => {
   847→   847→    return expression
   848→   848→      .split(/(\s+|×|÷|−|\+|=|\d+(?:\.\d+)?)/g)
   849→   849→      .filter((t) => t.trim() !== '');
   850→   850→  }, [expression]);
   851→   851→
   852→   852→  const resultStr = Number.isInteger(result) ? String(result) : result.toFixed(2);
   853→   853→
   854→   854→  return (
   855→   855→    <>
   856→   856→      {tokens.map((t, i) => {
   857→   857→        const isOp = ['+', '−', '×', '÷', '='].includes(t);
   858→   858→        return (
   859→   859→          <span
   860→   860→            key={i}
   861→   861→            className={
   862→   862→              isOp
   863→   863→                ? 'text-purple-500 px-1'
   864→   864→                : RAINBOW_COLORS[i % RAINBOW_COLORS.length]
   865→   865→            }
   866→   866→          >
   867→   867→            {t}
   868→   868→          </span>
   869→   869→        );
   870→   870→      })}
   871→   871→      <span className="text-purple-500 px-1">=</span>
   872→   872→      <motion.span
   873→   873→        initial={{ scale: 1.4 }}
   874→   874→        animate={{ scale: 1 }}
   875→   875→        transition={{ type: 'spring', stiffness: 300, damping: 10 }}
   876→   876→        className="text-green-600 px-2 underline decoration-wavy decoration-green-400"
   877→   877→      >
   878→   878→        {resultStr}
   879→   879→      </motion.span>
   880→   880→    </>
   881→   881→  );
   882→   882→}

  1017→  1017→// Deterministic PRNG (mulberry32) so the floating decorations are computed
  1018→  1018→// identically on the server and the client — no hydration mismatch, and
  1019→  1019→// no need to setState-after-mount.
  1020→  1020→function mulberry32(seed: number) {
  1021→  1021→  let a = seed >>> 0;
  1022→  1022→  return () => {
  1023→  1023→    a = (a + 0x6d2b79f5) >>> 0;
  1024→  1024→    let t = a;
  1025→  1025→    t = Math.imul(t ^ (t >>> 15), t | 1);
  1026→  1026→    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
  1027→  1027→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  1028→  1028→  };
  1029→  1029→}

  1102→  1102→  }
  1103→  1103→  if (n < 1_000_000) {
  1104→  1104→    const th = Math.floor(n / 1000);

  1107→  1107→  }
